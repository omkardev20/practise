    Array cheat sheet 
    // Common patterns:
let result = [];            // start fresh
result.push(item);          // add to end
result.includes(item);      // check if exists

// Manual transformation (like .map):
for (const item of arr) { result.push(item * 2); }

// Manual filtering (like .filter):
for (const item of arr) { if (item % 2 === 0) result.push(item); }

// Frequency Map (count items into an object):
let counts = {};
for (const item of arr) {
  counts[item] = (counts[item] || 0) + 1;
}

// Early exit:
for (const item of arr) {
  if (condition) return item; // stop early!
}
