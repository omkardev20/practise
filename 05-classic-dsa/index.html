    DSA Patterns Cheat Sheet 
    // 1. Two Pointers (opposite ends):
let left = 0, right = arr.length - 1;
while (left < right) {
  // compare or swap arr[left] and arr[right]
  left++; right--;
}

// 2. Frequency / Seen Map (O(n) lookups):
const seen = new Map(); // or const seen = {};
for (let i = 0; i < arr.length; i++) {
  const diff = target - arr[i];
  if (seen.has(diff)) return [seen.get(diff), i];
  seen.set(arr[i], i);
}

// 3. Stack (LIFO: push on open, pop on close):
const stack = [];
stack.push(char);
const top = stack.pop();

// 4. Binary Search (sorted array):
let low = 0, high = arr.length - 1;
while (low <= high) {
  const mid = Math.floor((low + high) / 2);
  if (arr[mid] === target) return mid;
  if (arr[mid] < target) low = mid + 1;
  else high = mid - 1;
}

// 5. Sliding Window (window sum / length):
let sum = 0, max = 0;
for (let i = 0; i < arr.length; i++) {
  sum += arr[i];
  if (i >= k) sum -= arr[i - k];
  max = Math.max(max, sum);
}
