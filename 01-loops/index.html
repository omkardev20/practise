    Loop cheat sheet 
    for (let i = 1; i <= 5; i++) { }    // count: start; keep going while true; step
for (let i = 5; i >= 1; i--) { }    // count backwards
for (const item of array) { }       // VALUES of an array or string
for (const key in object) { }       // KEYS of an object
while (n > 0) { n--; }              // when you don't know how many rounds

break;      // leave the loop
continue;   // skip to the next round
arr.push(x) // add x to the end of an array
a % b       // remainder: 7 % 2 is 1, so 7 is odd
